/**
 * Step 9f: the Garden map observes one garden (the first active
 * garden returned by the repository), plus its areas, growing spaces
 * and spatial objects, and exposes them as a single GardenMapUiState.
 *
 * The choice of "first garden" matches the POC and Q5's "one garden,
 * no in-map switcher" scope. Multi-garden selection is a Home-screen
 * concern and remains unimplemented.
 *
 * The ViewModel is where domain entities stop: it maps Garden /
 * Area / GrowingSpace / SpatialObject into MapShape values, which
 * carry only geometry plus a semantic kind. MapLibre never sees
 * domain models.
 */
@OptIn(ExperimentalCoroutinesApi::class)
@HiltViewModel
class GardenMapViewModel
    @Inject
    constructor(
        gardenRepository: GardenRepository,
        private val observeGrowingSpacesInGarden: ObserveGrowingSpacesInGarden,
        private val observeSpatialObjectsInGarden: ObserveSpatialObjectsInGarden,
        private val observeAreasInGarden: ObserveAreasInGarden,
    ) : ViewModel() {
        private val baseFlow: Flow<GardenMapUiState> =
            gardenRepository
                .observeGardens()
                .map { gardens -> gardens.firstOrNull() }
                .flatMapLatest { garden ->
                    if (garden == null) {
                        flowOf<GardenMapUiState>(GardenMapUiState.NoGarden)
                    } else {
                        gardenFlow(garden)
                    }
                }

        val uiState: StateFlow<GardenMapUiState> =
            baseFlow
                .catch { emit(GardenMapUiState.Error) }
                .stateIn(
                    scope = viewModelScope,
                    started = SharingStarted.WhileSubscribed(5_000),
                    initialValue = GardenMapUiState.Loading,
                )

        private fun gardenFlow(garden: Garden): Flow<GardenMapUiState> =
            combine(
                observeAreasInGarden(garden.id),
                observeSpatialObjectsInGarden(garden.id),
                observeGrowingSpacesInGarden(garden.id),
            ) { areas, objects, spaces ->
                buildState(garden, areas, objects, spaces)
            }

        private fun buildState(
            garden: Garden,
            areas: List<Area>,
            objects: List<SpatialObject>,
            spaces: List<GrowingSpace>,
        ): GardenMapUiState {
            val shapes = mutableListOf<MapShape>()
            areas.forEach { area ->
                val geometry = area.geometry ?: return@forEach
                shapes += MapShape(area.id, MapShapeKind.Area, geometry)
            }
            objects.forEach { obj ->
                shapes += MapShape(obj.id, MapShapeKind.SpatialObject(obj.objectType), obj.geometry)
            }
            spaces.forEach { space ->
                val geometry = space.geometry ?: return@forEach
                shapes += MapShape(space.id, MapShapeKind.GrowingSpace(space.spaceType), geometry)
            }
            if (shapes.isEmpty()) return GardenMapUiState.NoGeometries
            return GardenMapUiState.Ready(
                gardenName = garden.name,
                origin = originOf(garden),
                shapes = shapes,
                viewport = MapViewport.union(shapes),
            )
        }

        private fun originOf(garden: Garden): Wgs84 =
            Wgs84(
                latitude = garden.latitude ?: FALLBACK_LATITUDE,
                longitude = garden.longitude ?: FALLBACK_LONGITUDE,
            )

        private companion object {
            const val FALLBACK_LATITUDE = -43.5321
            const val FALLBACK_LONGITUDE = 172.6362
        }
    }

/**

@Composable
private fun rememberMapEditorSession(
    editState: GardenMapEditUiState,
    shapes: List<MapShape>,
): MapEditorSession {
    val context = LocalContext.current
    val session = remember(context) { MapEditorSession(MapView(context).also { it.onCreate(null) }) }
    LaunchedEffect(editState, shapes) {
        session.stateRef.value = editState
        session.shapesRef.value = shapes
    }
    val lifecycleOwner = LocalLifecycleOwner.current
    DisposableEffect(session.mapView, lifecycleOwner) {
        val observer =
            LifecycleEventObserver { _, event ->
                when (event) {
                    Lifecycle.Event.ON_START -> session.mapView.onStart()
                    Lifecycle.Event.ON_RESUME -> session.mapView.onResume()
                    Lifecycle.Event.ON_PAUSE -> session.mapView.onPause()
                    Lifecycle.Event.ON_STOP -> session.mapView.onStop()
                    else -> Unit
                }
            }
        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose {
            lifecycleOwner.lifecycle.removeObserver(observer)
            session.mapView.onDestroy()
        }
    }
    return session
}

                        mapView = mapView,
                        map = map,

@Composable
private fun ShapeLayersEffect(
    styleRef: Style?,
    styleVersion: Int,
    shapes: List<MapShape>,
    origin: Wgs84,
) {
    LaunchedEffect(styleVersion, shapes) {
        if (styleVersion == 0) return@LaunchedEffect
        val style = styleRef ?: return@LaunchedEffect
        style
            .getSourceAs<GeoJsonSource>(MapLibreStyleConstants.SOURCE_AREAS)
            ?.setGeoJson(GardenMapFeatures.areas(shapes, origin))
        style
            .getSourceAs<GeoJsonSource>(MapLibreStyleConstants.SOURCE_SPATIAL_OBJECTS)
            ?.setGeoJson(GardenMapFeatures.spatialObjects(shapes, origin))
        style
            .getSourceAs<GeoJsonSource>(MapLibreStyleConstants.SOURCE_GROWING_SPACES)
            ?.setGeoJson(GardenMapFeatures.growingSpaces(shapes, origin))
    }
}

@Composable
private fun SelectionLayersEffect(
    styleRef: Style?,
    styleVersion: Int,
    selectedShape: MapShape?,
    mode: MapEditMode,
    controller: MapEditorController,
    origin: Wgs84,
) {
    LaunchedEffect(
        styleVersion,
        selectedShape?.id,
        mode,
        controller.isDragging,
        controller.dragCurrentMeters,
    ) {
        if (styleVersion == 0) return@LaunchedEffect
        val style = styleRef ?: return@LaunchedEffect
        val geometry = controller.inProgressGeometry() ?: selectedShape?.geometry
        if (geometry != null) {
            style
                .getSourceAs<GeoJsonSource>(MapEditorLayers.SOURCE_SELECTION)
                ?.setGeoJson(MapEditorFeatures.selection(geometry, origin))
        } else {
            style
                .getSourceAs<GeoJsonSource>(MapEditorLayers.SOURCE_SELECTION)
                ?.setGeoJson(MapEditorFeatures.empty())
        }
        val showVertices = mode == MapEditMode.SELECT && !controller.isDragging && geometry is Geometry.Polygon
        val vertexCollection =
            if (showVertices) {
                MapEditorFeatures.vertices(geometry as Geometry.Polygon, origin)
            } else {
                MapEditorFeatures.empty()
            }
        style
            .getSourceAs<GeoJsonSource>(MapEditorLayers.SOURCE_VERTICES)
            ?.setGeoJson(vertexCollection)
    }
}

@Composable
private fun DrawPreviewEffect(
    styleRef: Style?,
    styleVersion: Int,
    drawState: MapDrawState,
    origin: Wgs84,
) {
    LaunchedEffect(styleVersion, drawState.vertices) {
        if (styleVersion == 0) return@LaunchedEffect
        val style = styleRef ?: return@LaunchedEffect
        val vertices = drawState.vertices
        val lineCollection =
            if (vertices.size >= 2) {
                MapEditorFeatures.drawPreview(vertices, origin)
            } else {
                MapEditorFeatures.empty()
            }
        style
            .getSourceAs<GeoJsonSource>(MapEditorLayers.SOURCE_DRAW_LINE)
            ?.setGeoJson(lineCollection)
        style
            .getSourceAs<GeoJsonSource>(MapEditorLayers.SOURCE_DRAW_VERTICES)
            ?.setGeoJson(MapEditorFeatures.drawVertices(vertices, origin))
    }
}

private const val FALLBACK_ZOOM = 17.0
private const val ZOOM_SCALE_BASE = 17.0
private const val MIN_ZOOM = 12.0
private const val MAX_ZOOM = 20.0

private fun fitViewport(
    map: MapLibreMap,
    viewport: MapViewport,
    origin: Wgs84,
) {
    val centreWgs =
        MetresToWgs84.toWgs84(
            origin,
            Coordinate(viewport.centerX, viewport.centerY),
        )
    val largestDimension = viewport.largestDimensionMetres
    val zoom =
        if (largestDimension <= 0.0) {
            FALLBACK_ZOOM
        } else {
            (ZOOM_SCALE_BASE - log2(largestDimension)).coerceIn(MIN_ZOOM, MAX_ZOOM)
        }
    map.cameraPosition =
        CameraPosition
            .Builder()
            .target(LatLng(centreWgs.latitude, centreWgs.longitude))
            .zoom(zoom)
            .build()
}

package com.mytuin.gardenplanner.ui.garden

import com.mytuin.gardenplanner.domain.model.garden.Coordinate
import com.mytuin.gardenplanner.domain.model.garden.Geometry
import org.maplibre.geojson.Feature
import org.maplibre.geojson.FeatureCollection
import org.maplibre.geojson.LineString
import org.maplibre.geojson.Point
import org.maplibre.geojson.Polygon

/**
 * FeatureCollection builders for the Garden map (Step 9f).
 *
 * Converts each MapShape into a MapLibre Feature with the style
 * properties that MapShapeStyle chose. Grouped by MapShapeKind so
 * each source receives only its own shapes and the layer stack keeps
 * the draw order the design intends (areas under, growing spaces on
 * top).
 *
 * The metre → WGS84 conversion happens here, using the garden
 * origin. It is the only place MapLibre types enter the codebase
 * alongside GardenMapStyle.
 */
object GardenMapFeatures {
    fun areas(
        shapes: List<MapShape>,
        origin: Wgs84,
    ): FeatureCollection = collection(shapes.filter { it.kind is MapShapeKind.Area }, origin)

    fun spatialObjects(
        shapes: List<MapShape>,
        origin: Wgs84,
    ): FeatureCollection = collection(shapes.filter { it.kind is MapShapeKind.SpatialObject }, origin)

    fun growingSpaces(
        shapes: List<MapShape>,
        origin: Wgs84,
    ): FeatureCollection = collection(shapes.filter { it.kind is MapShapeKind.GrowingSpace }, origin)

    private fun collection(
        shapes: List<MapShape>,
        origin: Wgs84,
    ): FeatureCollection {
        val features = shapes.mapNotNull { shape -> feature(shape, origin) }
        return FeatureCollection.fromFeatures(features)
    }

    private fun feature(
        shape: MapShape,
        origin: Wgs84,
    ): Feature? {
        val style = MapShapeStyle.forKind(shape.kind)
        val geometry =
            when (val g = shape.geometry) {
                is Geometry.Point -> point(g.at, origin)
                is Geometry.LineString -> line(g.vertices, origin)
                is Geometry.Polygon -> polygon(g.ring, origin)
            }
        return Feature
            .fromGeometry(geometry)
            .apply {
                addStringProperty(MapLibreStyleConstants.PROP_SHAPE_ID, shape.id)
                addStringProperty(MapLibreStyleConstants.PROP_FILL_COLOR, style.fillColor)
                addNumberProperty(MapLibreStyleConstants.PROP_FILL_OPACITY, style.fillOpacity.toDouble())
                addStringProperty(MapLibreStyleConstants.PROP_STROKE_COLOR, style.strokeColor)
                addNumberProperty(MapLibreStyleConstants.PROP_STROKE_WIDTH, style.strokeWidth.toDouble())
            }
    }

    private fun point(
        coordinate: Coordinate,
        origin: Wgs84,
    ): Point {
        val wgs = MetresToWgs84.toWgs84(origin, coordinate)
        return Point.fromLngLat(wgs.longitude, wgs.latitude)
    }

    private fun line(
        vertices: List<Coordinate>,
        origin: Wgs84,
    ): LineString {
        val points = vertices.map { point(it, origin) }
        return LineString.fromLngLats(points)
    }

    private fun polygon(
        ring: List<Coordinate>,
        origin: Wgs84,
    ): Polygon {
        val points = ring.map { point(it, origin) }
        return Polygon.fromLngLats(listOf(points))
    }
}

package com.mytuin.gardenplanner.ui.garden

import com.mytuin.gardenplanner.domain.model.garden.Geometry

/**
 * Bounding box in the garden's local metre frame (Step 9f).
 *
 * Used to fit the MapLibre camera so all shapes are visible on
 * first composition and when the user taps re-centre. The fallback
 * bounds give the camera somewhere sensible to point when a garden
 * exists but has no geometry yet.
 */
data class MapViewport(
    val minX: Double,
    val minY: Double,
    val maxX: Double,
    val maxY: Double,
) {
    val centerX: Double get() = (minX + maxX) / 2.0
    val centerY: Double get() = (minY + maxY) / 2.0
    val widthMetres: Double get() = maxX - minX
    val heightMetres: Double get() = maxY - minY
    val largestDimensionMetres: Double get() = maxOf(widthMetres, heightMetres)

    companion object {
        private const val PADDING_METRES = 3.0
        private const val FALLBACK_HALF_EXTENT = 15.0

        val FALLBACK: MapViewport =
            MapViewport(
                minX = -FALLBACK_HALF_EXTENT,
                minY = -FALLBACK_HALF_EXTENT,
                maxX = FALLBACK_HALF_EXTENT,
                maxY = FALLBACK_HALF_EXTENT,
            )

        /**
         * Union of all coordinates across every shape, expanded by
         * PADDING_METRES on each side so shapes are not flush with
         * the screen edge. Returns FALLBACK when shapes contain no
         * coordinates.
         */
        fun union(shapes: List<MapShape>): MapViewport {
            val coords = shapes.flatMap { shape -> shape.geometry.allCoordinates() }
            if (coords.isEmpty()) return FALLBACK
            val xs = coords.map { it.x }
            val ys = coords.map { it.y }
            return MapViewport(
                minX = xs.min() - PADDING_METRES,
                minY = ys.min() - PADDING_METRES,
                maxX = xs.max() + PADDING_METRES,
                maxY = ys.max() + PADDING_METRES,
            )
        }

        private fun Geometry.allCoordinates(): List<com.mytuin.gardenplanner.domain.model.garden.Coordinate> =
            when (this) {
                is Geometry.Point -> listOf(at)
                is Geometry.LineString -> vertices
                is Geometry.Polygon -> ring
            }
    }
}

package com.mytuin.gardenplanner.ui.theme

import android.app.Activity
import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.darkColorScheme
import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable
import androidx.compose.runtime.SideEffect
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.platform.LocalView
import androidx.core.view.WindowCompat

private val LightColorScheme =
    lightColorScheme(
        primary = Forest,
        onPrimary = Color.White,
        primaryContainer = Sage,
        onPrimaryContainer = DeepForest,
        secondary = Sage,
        onSecondary = Color.White,
        background = WarmWhite,
        onBackground = Charcoal,
        surface = WarmWhite,
        onSurface = Charcoal,
        surfaceVariant = Paper,
        onSurfaceVariant = Charcoal,
        outline = Stone,
        outlineVariant = Stone,
        error = Clay,
        onError = Color.White,
    )

private val DarkColorScheme =
    darkColorScheme(
        primary = Sage,
        onPrimary = DeepForest,
        primaryContainer = DeepForest,
        onPrimaryContainer = Sage,
        secondary = Sage,
        onSecondary = DeepForest,
        background = DarkSurface,
        onBackground = WarmWhite,
        surface = DarkSurface,
        onSurface = WarmWhite,
        surfaceVariant = DarkSurfaceVariant,
        onSurfaceVariant = WarmWhite,
        outline = MidStone,
        outlineVariant = MidStone,
        error = Clay,
        onError = Color.White,
    )

/**
 * V1_VISUAL_DESIGN_SPECIFICATION §53–§55: both themes supported.
 * Light is the default; dark follows the system.
 */
@Composable
fun MyTuinTheme(
    darkTheme: Boolean = isSystemInDarkTheme(),
    content: @Composable () -> Unit,
) {
    val colorScheme = if (darkTheme) DarkColorScheme else LightColorScheme

    val view = LocalView.current
    if (!view.isInEditMode) {
        SideEffect {
            val window = (view.context as Activity).window
            WindowCompat
                .getInsetsController(window, view)
                .isAppearanceLightStatusBars = !darkTheme
        }
    }

    MaterialTheme(
        colorScheme = colorScheme,
        typography = MyTuinTypography,
        content = content,
    )
}
